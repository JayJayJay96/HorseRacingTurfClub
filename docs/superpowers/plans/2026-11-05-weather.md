# Weather Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Races can run in Rain (3-lane puddles, 15 % slip) or Fog (neighbour bumps), rolled at odds lock, priced into the odds, shown on every screen, with an office on/off switch and HD-2D puddle reflections.

**Architecture:** Weather is a new optional `RaceInput.weather`; the sim draws extra randomness only when it is not clear, so clear races and old replays are bit-identical. The `lock-odds` Edge Function rolls it, prices the odds with it and stores it on `races.weather`; `start-race` reads it back. Both renderers draw puddles, rain and fog from `timeline.weather` / `timeline.puddles`, and splash/bonk effects from new `slip` / `bump` events.

**Tech Stack:** TypeScript strict (`.ts` imports, `noUncheckedIndexedAccess`), Vitest, Supabase (Postgres migrations, Deno Edge Functions bundling `src/sim`), React 19, Three.js 0.187 (HD-2D), Canvas 2D.

**Spec:** `docs/superpowers/specs/2026-11-05-weather-design.md`

**Commands:** `npm test` (unit), `npm run typecheck`, `npm run lint`, `npm run test:db` (needs `npx supabase start`), `npm run build`.
```ts
import { BOX_ZONE, WEATHER, WEATHER_WEIGHTS } from './config.ts';
import { round2 } from './math.ts';
import type { Rng } from './rng.ts';
import type { BoxSpot, Puddle, Weather } from './types.ts';
import { pickOne, pickWeighted } from './weighted.ts';

/** The race's weather, rolled at odds lock (spec 2026-11-05). Off = always clear. */
export function rollWeather(rng: Rng, enabled: boolean): Weather {
  return enabled ? pickWeighted(rng, WEATHER_WEIGHTS) : 'clear';
}

/**
 * Wide puddles for a rain race. Each covers `WEATHER.puddleLanes` adjacent lanes (all lanes when
 * there are fewer). Every puddle goes over the least-covered lane, in the window round it with the
 * fewest puddles, so edge lanes get as many as the rest. `boxes` is per lane (lane = index).
 * A puddle that finds no free spot in `placeTries` draws is dropped.
 */
export function placePuddles(
  rng: Rng,
  lanes: number,
  boxes: readonly (readonly BoxSpot[])[],
  distance: number,
  perLane: number,
): Puddle[] {
  const span = Math.min(WEATHER.puddleLanes, lanes);
  const total = Math.ceil((lanes * perLane) / span);
  const len = WEATHER.puddleLengthM;
  const from = distance * BOX_ZONE.start;
  const room = distance * (BOX_ZONE.end - BOX_ZONE.start) - len;
  const count = new Array<number>(lanes).fill(0);
  const out: Puddle[] = [];
  const windowSum = (s: number) => count.slice(s, s + span).reduce((a, b) => a + b, 0);
  for (let k = 0; k < total; k++) {
    const least = Math.min(...count);
    const lane = pickOne(
      rng,
      count.flatMap((c, i) => (c === least ? [i] : [])),
    );
    const starts: number[] = [];
    for (let s = Math.max(0, lane - span + 1); s <= Math.min(lane, lanes - span); s++) starts.push(s);
    const best = Math.min(...starts.map(windowSum));
    const s0 = pickOne(
      rng,
      starts.filter((s) => windowSum(s) === best),
    );
    const covered: [number, number] = [s0, s0 + span - 1];
    for (let tries = 0; tries < WEATHER.placeTries; tries++) {
      const at = round2(from + (rng.nextInt(1_000_001) / 1_000_000) * room);
      const clash = (l: number) =>
        boxes[l]!.some((b) => b.at >= at - WEATHER.boxClearM && b.at <= at + len + WEATHER.boxClearM) ||
        out.some((p) => p.lanes[0] <= l && l <= p.lanes[1] && p.at < at + len && at < p.at + len);
      let free = true;
      for (let l = covered[0]; l <= covered[1]; l++) if (clash(l)) free = false;
      if (!free) continue;
      out.push({ at, lanes: covered });
      for (let l = covered[0]; l <= covered[1]; l++) count[l]!++;
      break;
    }
  }
  return out.sort((a, b) => a.at - b.at);
}

/** Each lane's puddles, sorted by distance. */
export function puddlesByLane(puddles: readonly Puddle[], lanes: number): Puddle[][] {
  return Array.from({ length: lanes }, (_, lane) =>
    puddles.filter((p) => p.lanes[0] <= lane && lane <= p.lanes[1]),
  );
}
```

Lane 0 only belongs to the window starting at 0, so it is picked first whenever it is least covered; the per-lane spread stays within 2.

```ts
import { WEATHER } from '../sim/config.ts';
import type { Puddle } from '../sim/types.ts';
import { finishS, trackPoint, type Point, type Track } from './track.ts';

/** Light touch (owner 2026-11-05): vibe only, horses always readable. */
export const RAIN = {
  /** Streaks in a box this many metres round the camera target, up to `top` high. */
  drops: 260,
  half: 45,
  top: 30,
  /** Fall speed, m/s. */
  speed: 28,
  /** Streak length (m) and its sideways lean per metre of fall. */
  length: 1.2,
  lean: 0.25,
  /** Multiplies the HD post grade; 2D draws its streaks at `alpha2d`. */
  tint: [0.86, 0.9, 0.97] as const,
  saturation: 0.85,
  alpha2d: 0.22,
} as const;

export const FOG = {
  /** HD: fog starts this many shot-distances out (clear uses 1.2 / 4). */
  near: 1.15,
  far: 2.6,
  /** HD fog colour mixes this much toward `color`; 2D haze peaks at `alpha2d`. */
  mix: 0.55,
  color: '#e3e8ec',
  alpha2d: 0.35,
} as const;

/** Deterministic 0..1 noise per index (same recipe as effects.ts). */
const hash = (i: number, salt: number) => {
  const x = Math.sin(i * 12.9898 + salt * 78.233) * 43758.5453;
  return x - Math.floor(x);
};

/**
 * World-space outline of a puddle: an ellipse over its lanes and its length, with a small
 * per-puddle wobble so they don't look stamped.
 */
export function puddleOutline(track: Track, distance: number, puddle: Puddle, points = 16): Point[] {
  const len = WEATHER.puddleLengthM;
  const sMid = finishS(track) - distance + puddle.at + len / 2;
  const lw = track.laneWidth;
  const inner = puddle.lanes[0] * lw;
  const outer = (puddle.lanes[1] + 1) * lw;
  const mid = (inner + outer) / 2;
  const half = (outer - inner) / 2 - 0.2;
  return Array.from({ length: points }, (_, i) => {
    const a = (i / points) * 2 * Math.PI;
    const wob = 1 + (hash(i, puddle.at) - 0.5) * 0.18;
    return trackPoint(track, sMid + Math.cos(a) * (len / 2) * wob, mid + Math.sin(a) * half * wob);
  });
}

/** A rain streak's top end relative to the camera target (y up), looping every top/speed s. */
export function rainDrop(i: number, t: number): { x: number; y: number; z: number } {
  const x = (hash(i, 1) * 2 - 1) * RAIN.half;
  const z = (hash(i, 2) * 2 - 1) * RAIN.half;
  const phase = hash(i, 3) * RAIN.top;
  const y = RAIN.top - ((phase + t * RAIN.speed) % RAIN.top);
  return { x, y, z };
}
```
