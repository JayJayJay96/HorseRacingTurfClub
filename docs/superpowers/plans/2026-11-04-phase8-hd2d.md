# Phase 8: HD-2D Race Renderer — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Three.js renderer that plays the existing race `Timeline` in an HD-2D (Octopath-Traveler-like) look on every screen that uses `RaceView`, with the pixel 2D renderer as fallback and manual choice.

**Architecture:** Pure, unit-tested modules (`look`, `lighting`, `textures`, `sheet`, `scenery`, `shots`, `effects`, `ambient`, `renderSetting`) hold every number and rule; thin Three.js modules (`scene`, `post`, `HD2DRenderer`) turn them into a lit diorama with billboard sprites, sun shadows and a post stack (tilt-shift → bloom → grade + vignette → output). `RaceView` lazy-loads the HD renderer with `import()`, so phones and the 2D path never download Three.js.

**Tech Stack:** TypeScript (strict, `noUncheckedIndexedAccess`, `erasableSyntaxOnly`, `.ts` import extensions), React 19, Three.js r187 (`three@0.187.0`, addons from `three/addons/...`), Vitest (node environment), Playwright.

**Spec:** `docs/superpowers/specs/2026-11-04-hd2d-renderer-design.md`. Its **Look targets** win over looser wording. Task 9 is the owner's look-test gate: stop there until the owner approves.

```ts
/** Per-browser race graphics choice (a convenience, never game state). */
export type RenderMode = 'hd' | '2d';

export const RENDER_KEY = 'hrtc.render';

function safeStorage(): Storage | null {
  try {
    return window.localStorage;
  } catch {
    return null;
  }
}

export function loadRenderMode(storage: Storage | null = safeStorage()): RenderMode {
  try {
    return storage?.getItem(RENDER_KEY) === '2d' ? '2d' : 'hd';
  } catch {
    return 'hd';
  }
}

export function saveRenderMode(mode: RenderMode, storage: Storage | null = safeStorage()): boolean {
  if (!storage) return false;
  try {
    storage.setItem(RENDER_KEY, mode);
    return true;
  } catch {
    return false;
  }
}

/** HD only when the viewer wants it, the browser has WebGL2 and HD hasn't failed on this page. */
export const pickRenderer = (mode: RenderMode, webgl2: boolean, failed: boolean): RenderMode =>
  mode === 'hd' && webgl2 && !failed ? 'hd' : '2d';

/** Probes a throwaway canvas and releases its context at once. */
export function webgl2Available(): boolean {
  try {
    const gl = document.createElement('canvas').getContext('webgl2');
    gl?.getExtension('WEBGL_lose_context')?.loseContext();
    return gl !== null;
  } catch {
    return false;
  }
}
```

```ts
/**
 * Look targets (spec "Look targets"). Task 9's look test may change these numbers; once the owner
 * approves, they are copied into the spec and frozen.
 */

/** World size of one sprite or texture pixel, metres (Look target 2). */
export const PIXEL_M = 0.15;
/** Screen pixels per sprite pixel on the chase shot at REFERENCE_HEIGHT (Look target 2). */
export const PIXEL_SCALE = 4.5;
export const REFERENCE_HEIGHT = 1080;
/** Vertical field of view, degrees (Look target 3). */
export const FOV_DEG = 25;
/** Ground texture tile, in pixels and metres. */
export const TILE_PX = 32;
export const TILE_M = TILE_PX * PIXEL_M;
/** Camera pitch below horizontal per shot, degrees (Look target 3). */
export const PITCH_DEG = { wide: 50, chase: 35, finish: 28 } as const;
/** Finish shot sits this much farther than the chase shot. */
export const FINISH_DISTANCE_SCALE = 1.3;
/** Tilt-shift blur per shot, in screen pixels at the band edges. */
export const TILT_BLUR_PX = { wide: 4, chase: 2, finish: 2 } as const;
/** Where the sharp band sits, as a share of the screen height from the bottom. */
export const TILT_FOCUS = 0.45;
/** Per-frame ease toward the shot target (same as the 2D camera). */
export const EASE = 0.06;

/** Camera distance at which one PIXEL_M covers `pixelScale` screen px at `viewH`. */
export function distanceForScale(
  pixelScale: number = PIXEL_SCALE,
  fovDeg: number = FOV_DEG,
  viewH: number = REFERENCE_HEIGHT,
): number {
  const pxPerMetre = pixelScale / PIXEL_M;
  return viewH / (2 * Math.tan((fovDeg * Math.PI) / 360) * pxPerMetre);
}
```

```ts
import { createSeededRng } from '../../sim/rng.ts';
import { TILE_PX } from './look.ts';

/** RGBA pixels, bottom row first (GL order). */
export interface PixelImage {
  readonly width: number;
  readonly height: number;
  readonly data: Uint8ClampedArray;
}

export type TileKind = 'grass' | 'dirt' | 'wood' | 'crowd';
export type SceneryKind = 'tree' | 'bush' | 'flag';

export const PALETTES: Readonly<Record<TileKind, readonly string[]>> = {
  grass: ['#3d7a3a', '#356b33', '#4a8a43', '#5c9e4c'],
  dirt: ['#b98a5a', '#a97b4e', '#c79a68', '#8f6740'],
  wood: ['#7a5a3a', '#6a4c30', '#8c6a46', '#4e3824'],
  // seats, two skin tones, then shirts
  crowd: ['#4e4e5e', '#e0b48a', '#8d5a3b', '#e53935', '#1e88e5', '#fdd835', '#43a047', '#ffffff'],
};

const rgb = (hex: string): [number, number, number] => [
  parseInt(hex.slice(1, 3), 16),
  parseInt(hex.slice(3, 5), 16),
  parseInt(hex.slice(5, 7), 16),
];

function image(width: number, height: number) {
  const data = new Uint8ClampedArray(width * height * 4);
  /** (x, y) with y from the TOP, so drawing code reads naturally; stored bottom row first. */
  const set = (x: number, y: number, hex: string) => {
    if (x < 0 || y < 0 || x >= width || y >= height) return;
    const i = ((height - 1 - y) * width + x) * 4;
    const [r, g, b] = rgb(hex);
    data[i] = r;
    data[i + 1] = g;
    data[i + 2] = b;
    data[i + 3] = 255;
  };
  const img: PixelImage = { width, height, data };
  return { img, set };
}

/** Pick an index with integer weights that sum to 100. */
function weighted(roll: number, weights: readonly number[]): number {
  let acc = 0;
  for (let i = 0; i < weights.length; i++) {
    acc += weights[i]!;
    if (roll < acc) return i;
  }
  return weights.length - 1;
}

/** A tiling pixel-art texture. */
export function pixelTexture(
  kind: TileKind,
  seed: number,
  width: number = TILE_PX,
  height: number = TILE_PX,
): PixelImage {
  const rng = createSeededRng(seed);
  const pal = PALETTES[kind];
  const { img, set } = image(width, height);
  for (let y = 0; y < height; y++) {
    for (let x = 0; x < width; x++) {
      const roll = rng.nextInt(100);
      if (kind === 'grass') set(x, y, pal[weighted(roll, [60, 25, 12, 3])]!);
      else if (kind === 'dirt') set(x, y, pal[weighted(roll, [55, 25, 15, 5])]!);
      else if (kind === 'wood') {
        // 4 px planks with a dark seam, light grain flecks.
        const seam = y % 4 === 3;
        set(x, y, seam ? pal[3]! : roll < 10 ? pal[2]! : pal[(y >> 2) % 2]!);
      } else set(x, y, pal[0]!);
    }
  }
  if (kind === 'crowd') {
    // 2-wide people every 4 px, rows of 8 px: head 2x2, body 2x3.
    for (let row = 0; row * 8 < height; row++) {
      for (let col = 0; col * 4 < width; col++) {
        if (rng.nextInt(10) === 0) continue; // an empty seat
        const x = col * 4 + 1;
        const y = row * 8 + 2;
        const skin = pal[1 + rng.nextInt(2)]!;
        const shirt = pal[3 + rng.nextInt(5)]!;
        for (let dx = 0; dx < 2; dx++) {
          for (let dy = 0; dy < 2; dy++) set(x + dx, y + dy, skin);
          for (let dy = 2; dy < 5; dy++) set(x + dx, y + dy, shirt);
        }
      }
    }
  }
  return img;
}

/** Pixel-art scenery cut-outs with transparent backgrounds. */
export function scenerySprite(kind: SceneryKind, seed: number): PixelImage {
  const rng = createSeededRng(seed);
  if (kind === 'tree') {
    const { img, set } = image(24, 32);
    for (let y = 22; y < 32; y++) for (let x = 10; x < 14; x++) set(x, y, '#5a3b22');
    const blobs = [
      [12, 10, 9],
      [7, 15, 6],
      [17, 15, 6],
    ] as const;
    for (let y = 0; y < 24; y++) {
      for (let x = 1; x < 23; x++) {
        const inside = blobs.some(([cx, cy, r]) => (x - cx) ** 2 + (y - cy) ** 2 <= r * r);
        if (!inside) continue;
        // Light from the top-left: lighter up there, darker bottom-right, a few flecks.
        const shade = x + y < 18 ? '#5c9e4c' : x + y > 30 ? '#24521f' : '#3d7a3a';
        set(x, y, rng.nextInt(12) === 0 ? '#4a8a43' : shade);
      }
    }
    return img;
  }
  if (kind === 'bush') {
    const { img, set } = image(16, 10);
    for (let y = 1; y < 10; y++) {
      for (let x = 1; x < 15; x++) {
        if (((x - 7.5) / 7) ** 2 + ((y - 6) / 5) ** 2 > 1) continue;
        set(x, y, y < 5 ? '#4a8a43' : rng.nextInt(6) === 0 ? '#5c9e4c' : '#356b33');
      }
    }
    return img;
  }
  const { img, set } = image(10, 24);
  for (let y = 2; y < 24; y++) set(1, y, '#dddddd');
  const cloth = ['#e53935', '#1e88e5', '#fdd835'][rng.nextInt(3)]!;
  for (let y = 2; y < 8; y++) for (let x = 2; x < 9; x++) set(x, y, (x + y) % 5 ? cloth : '#ffffff');
  return img;
}

/** 1 px wide vertical gradient: `top` colour at the top, `bottom` at the bottom. */
export function gradient(top: string, bottom: string, height: number): PixelImage {
  const { img, set } = image(1, height);
  const a = rgb(top);
  const b = rgb(bottom);
  for (let y = 0; y < height; y++) {
    const f = height === 1 ? 0 : y / (height - 1);
    const c = a.map((v, i) => Math.round(v + (b[i]! - v) * f));
    set(0, y, '#' + c.map((v) => v.toString(16).padStart(2, '0')).join(''));
  }
  return img;
}
```

```ts
import type { HorseColors } from '../palette.ts';
import { drawHorse, type Pose } from '../sprites.ts';

/** One cell fits every pose of `drawHorse` at size 1; hooves at (footX, footY) from the top-left. */
export const CELL = { w: 24, h: 22, footX: 12, footY: 21 } as const;
export const POSES: readonly Pose[] = ['side-right', 'side-left', 'away', 'toward'];
export const FRAMES = 4;

export const sheetSize = (lanes: number) => ({
  width: POSES.length * FRAMES * CELL.w,
  height: lanes * CELL.h,
});

export function cellOf(lane: number, pose: Pose, frame: number) {
  const f = ((frame % FRAMES) + FRAMES) % FRAMES;
  return { col: POSES.indexOf(pose) * FRAMES + f, row: lane };
}

/** Texture offset/repeat for a cell of a canvas texture (flipY = true, so row 0 is at the top). */
export function cellUv(lanes: number, cell: { col: number; row: number }) {
  const { width, height } = sheetSize(lanes);
  return {
    offsetX: (cell.col * CELL.w) / width,
    offsetY: 1 - ((cell.row + 1) * CELL.h) / height,
    repeatX: CELL.w / width,
    repeatY: CELL.h / height,
  };
}

/** Draws every lane's poses and gallop frames with the 2D sprite code. */
export function drawSheet(ctx: CanvasRenderingContext2D, colors: readonly HorseColors[]) {
  ctx.imageSmoothingEnabled = false;
  colors.forEach((c, lane) => {
    for (const pose of POSES) {
      for (let f = 0; f < FRAMES; f++) {
        const { col, row } = cellOf(lane, pose, f);
        drawHorse(ctx, col * CELL.w + CELL.footX, row * CELL.h + CELL.footY, c, f, pose, 1);
      }
    }
  });
}
```

```ts
import { createSeededRng } from '../../sim/rng.ts';
import { trackBounds, type Track } from '../track.ts';
import type { SceneryKind } from './textures.ts';

export interface SceneryItem {
  readonly kind: SceneryKind;
  /** Track metres. */
  readonly x: number;
  readonly y: number;
  /** Picks the sprite variant (0-3). */
  readonly variant: number;
}

/** Metres of clear ground kept around the track's bounding box. */
export const SCENERY_MARGIN = 12;
const BAND = 80;

/** The stands as the 2D renderer places them (pushed out by the field's width), plus 4 m. */
export function standsRect(track: Track, lanes: number) {
  const outer = lanes * track.laneWidth;
  const { stands } = track;
  return {
    x0: stands.x0 - 4,
    y0: stands.y0 + outer - 4,
    x1: stands.x1 + 4,
    y1: stands.y1 + outer + 4,
  };
}

/** Trees, bushes and flags in a band around the track, the same for every viewer of a race. */
export function placeScenery(
  track: Track,
  lanes: number,
  seed: number,
  count = 90,
): SceneryItem[] {
  const rng = createSeededRng(seed ^ 0x5eed);
  const { halfWidth: hw, halfHeight: hh } = trackBounds(track, lanes);
  const s = standsRect(track, lanes);
  const out: SceneryItem[] = [];
  const span = (half: number) => 2 * (half + SCENERY_MARGIN + BAND);
  for (let tries = 0; out.length < count && tries < count * 50; tries++) {
    const x = (rng.nextInt(10_000) / 10_000 - 0.5) * span(hw);
    const y = (rng.nextInt(10_000) / 10_000 - 0.5) * span(hh);
    const offTrack = Math.abs(x) > hw + SCENERY_MARGIN || Math.abs(y) > hh + SCENERY_MARGIN;
    const inStands = x > s.x0 && x < s.x1 && y > s.y0 && y < s.y1;
    if (!offTrack || inStands) continue;
    const roll = rng.nextInt(100);
    const kind: SceneryKind = roll < 50 ? 'tree' : roll < 85 ? 'bush' : 'flag';
    out.push({ kind, x, y, variant: rng.nextInt(4) });
  }
  return out;
}
```

```ts
import * as THREE from 'three';
import { finishS, trackPoint, type Track } from '../track.ts';
import type { Preset } from './lighting.ts';
import { PIXEL_M, TILE_M } from './look.ts';
import { placeScenery, standsRect } from './scenery.ts';

/** A pixel-sharp texture from a bottom-row-first image. */
export function toTexture(img: PixelImage, repeat: boolean): THREE.DataTexture {
  const tex = new THREE.DataTexture(img.data, img.width, img.height, THREE.RGBAFormat);
  tex.magFilter = THREE.NearestFilter;
  tex.minFilter = THREE.NearestFilter;
  tex.generateMipmaps = false;
  tex.colorSpace = THREE.SRGBColorSpace;
  if (repeat) tex.wrapS = tex.wrapT = THREE.RepeatWrapping;
  tex.needsUpdate = true;
  return tex;
}

/** Lit, alpha-tested billboard material whose shadow keeps the cut-out shape. */
export function cutoutMaterials(map: THREE.Texture) {
  const material = new THREE.MeshLambertMaterial({ map, alphaTest: 0.5, side: THREE.DoubleSide });
  const depth = new THREE.MeshDepthMaterial({
    depthPacking: THREE.RGBADepthPacking,
    map,
    alphaTest: 0.5,
  });
  return { material, depth };
}

/** A flat strip along the track between two offsets, uv in TILE_M units. */
function strip(track: Track, from: number, to: number, y: number, s0 = 0, s1 = track.lap) {
  const pos: number[] = [];
  const uv: number[] = [];
  const index: number[] = [];
  const step = Math.min(2, (s1 - s0) / 2);
  let n = 0;
  for (let s = s0; s <= s1 + 1e-6; s += step, n++) {
    for (const off of [from, to]) {
      const p = trackPoint(track, s, off);
      pos.push(p.x, y, p.y);
      uv.push(s / TILE_M, off / TILE_M);
    }
    if (n > 0) {
      const a = (n - 1) * 2;
      index.push(a, a + 1, a + 2, a + 1, a + 3, a + 2);
    }
  }
  const g = new THREE.BufferGeometry();
  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
  g.setAttribute('uv', new THREE.Float32BufferAttribute(uv, 2));
  g.setIndex(index);
  g.computeVertexNormals();
  return g;
}

/** A vertical band (the rail) along the track at `offset`, between heights y0 and y1. */
function band(track: Track, offset: number, y0: number, y1: number) {
  const pos: number[] = [];
  const index: number[] = [];
  let n = 0;
  for (let s = 0; s <= track.lap + 1e-6; s += 2, n++) {
    const p = trackPoint(track, s, offset);
    pos.push(p.x, y0, p.y, p.x, y1, p.y);
    if (n > 0) {
      const a = (n - 1) * 2;
      index.push(a, a + 1, a + 2, a + 1, a + 3, a + 2);
    }
  }
  const g = new THREE.BufferGeometry();
  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
  g.setIndex(index);
  g.computeVertexNormals();
  return g;
}

export interface World {
  readonly group: THREE.Group;
  /** Crowd rows, bobbed by the renderer; `userData.baseY` holds the rest height. */
  readonly crowd: THREE.Mesh[];
  /** Everything that turns to face the camera's yaw each frame. */
  readonly billboards: THREE.Object3D[];
  /** Top of the finish post, world coords (for confetti). */
  readonly finishTop: THREE.Vector3;
  readonly sky: THREE.DataTexture;
}

export function buildWorld(track: Track, lanes: number, seed: number, preset: Preset): World {
  const group = new THREE.Group();
  const outer = lanes * track.laneWidth;
  const billboards: THREE.Object3D[] = [];
  const crowd: THREE.Mesh[] = [];

  // Ground: grass everywhere.
  const size = 2400;
  const grassTex = toTexture(pixelTexture('grass', seed), true);
  grassTex.repeat.set(size / TILE_M, size / TILE_M);
  const ground = new THREE.Mesh(
    new THREE.PlaneGeometry(size, size).rotateX(-Math.PI / 2),
    new THREE.MeshLambertMaterial({ map: grassTex }),
  );
  ground.receiveShadow = true;
  group.add(ground);

  // Dirt track across all lanes.
  const dirt = new THREE.Mesh(
    strip(track, -2, outer + 2, 0.02),
    new THREE.MeshLambertMaterial({ map: toTexture(pixelTexture('dirt', seed + 1), true) }),
  );
  dirt.receiveShadow = true;
  group.add(dirt);

  // White rails (inner and outer) with posts every 6 m.
  const railMat = new THREE.MeshLambertMaterial({ color: '#f5f5f5', side: THREE.DoubleSide });
  const postGeo = new THREE.BoxGeometry(0.12, 1, 0.12);
  for (const offset of [-2.2, outer + 2.2]) {
    const rail = new THREE.Mesh(band(track, offset, 0.85, 1.0), railMat);
    rail.castShadow = true;
    group.add(rail);
    const count = Math.floor(track.lap / 6);
    const posts = new THREE.InstancedMesh(postGeo, railMat, count);
    const m = new THREE.Matrix4();
    for (let i = 0; i < count; i++) {
      const p = trackPoint(track, i * 6, offset);
      posts.setMatrixAt(i, m.makeTranslation(p.x, 0.5, p.y));
    }
    posts.castShadow = true;
    group.add(posts);
  }

  // Finish line and post.
  const fs = finishS(track);
  group.add(
    new THREE.Mesh(
      strip(track, -2, outer + 2, 0.03, fs - 0.3, fs + 0.3),
      new THREE.MeshLambertMaterial({ color: '#ffffff' }),
    ),
  );
  const postAt = trackPoint(track, fs, outer + 3);
  const post = new THREE.Mesh(
    new THREE.BoxGeometry(0.3, 6, 0.3),
    new THREE.MeshLambertMaterial({ color: '#ffffff' }),
  );
  post.position.set(postAt.x, 3, postAt.y);
  post.castShadow = true;
  const board = new THREE.Mesh(
    new THREE.BoxGeometry(2, 1.2, 0.2),
    new THREE.MeshLambertMaterial({ color: '#e53935' }),
  );
  board.position.set(postAt.x, 6.4, postAt.y);
  board.castShadow = true;
  group.add(post, board);

  // Stands: 4 stepped wooden tiers, a crowd row on each, facing the track (-z).
  const s = standsRect(track, lanes);
  const width = s.x1 - s.x0 - 8;
  const cx = (s.x0 + s.x1) / 2;
  const woodTex = toTexture(pixelTexture('wood', seed + 2), true);
  woodTex.repeat.set(width / TILE_M, 1);
  const woodMat = new THREE.MeshLambertMaterial({ map: woodTex });
  const crowdTex = toTexture(pixelTexture('crowd', seed + 3, 32, 16), true);
  const crowdH = 16 * PIXEL_M;
  crowdTex.repeat.set(width / TILE_M, 1);
  const { material: crowdMat } = cutoutMaterials(crowdTex);
  for (let i = 0; i < 4; i++) {
    const h = 1.2 * (i + 1);
    const tier = new THREE.Mesh(new THREE.BoxGeometry(width, h, 6), woodMat);
    const z = s.y0 + 4 + 3 + i * 6;
    tier.position.set(cx, h / 2, z);
    tier.castShadow = tier.receiveShadow = true;
    group.add(tier);
    const row = new THREE.Mesh(new THREE.PlaneGeometry(width, crowdH).rotateY(Math.PI), crowdMat);
    row.position.set(cx, h + crowdH / 2, z - 1);
    row.userData.baseY = row.position.y;
    group.add(row);
    crowd.push(row);
  }

  // Scenery cut-outs: 4 variants per kind, shared textures.
  const looks = new Map<string, ReturnType<typeof cutoutMaterials> & { w: number; h: number }>();
  for (const item of placeScenery(track, lanes, seed)) {
    const key = `${item.kind}:${item.variant}`;
    let look = looks.get(key);
    if (!look) {
      const img = scenerySprite(item.kind, seed + item.variant);
      look = { ...cutoutMaterials(toTexture(img, false)), w: img.width, h: img.height };
      looks.set(key, look);
    }
    const geo = new THREE.PlaneGeometry(look.w * PIXEL_M, look.h * PIXEL_M);
    geo.translate(0, (look.h * PIXEL_M) / 2, 0);
    const mesh = new THREE.Mesh(geo, look.material);
    mesh.customDepthMaterial = look.depth;
    mesh.castShadow = true;
    mesh.position.set(item.x, 0, item.y);
    group.add(mesh);
    billboards.push(mesh);
  }

  // Night: four floodlight towers outside the track's corners, glowing heads (bloom).
  if (preset.floodlights) {
    const head = new THREE.MeshBasicMaterial({
      color: new THREE.Color('#fff8e0').multiplyScalar(3),
    });
    const pole = new THREE.MeshLambertMaterial({ color: '#888888' });
    const corner = trackPoint(track, 0, outer + 30);
    for (const [sx, sz] of [
      [1, 1],
      [1, -1],
      [-1, 1],
      [-1, -1],
    ] as const) {
      const x = sx * (Math.abs(corner.x) + 60);
      const z = sz * (Math.abs(corner.y) + 20);
      const tower = new THREE.Mesh(new THREE.BoxGeometry(0.8, 28, 0.8), pole);
      tower.position.set(x, 14, z);
      const lamp = new THREE.Mesh(new THREE.BoxGeometry(5, 2, 1), head);
      lamp.position.set(x, 28, z);
      group.add(tower, lamp);
    }
  }

  // Sunset: soft additive light shafts in front of the stands.
  if (preset.shafts) {
    const shaftTex = toTexture(gradient('#000000', '#ffffff', 32), false);
    shaftTex.magFilter = shaftTex.minFilter = THREE.LinearFilter;
    const shaftMat = new THREE.MeshBasicMaterial({
      map: shaftTex,
      color: '#ffb070',
      transparent: true,
      opacity: 0.25,
      blending: THREE.AdditiveBlending,
      depthWrite: false,
      side: THREE.DoubleSide,
    });
    for (let i = 0; i < 4; i++) {
      const shaft = new THREE.Mesh(new THREE.PlaneGeometry(6, 40), shaftMat);
      shaft.position.set(cx - width / 3 + (i * width) / 4.5, 20, s.y0 - 6);
      shaft.rotation.z = -0.35;
      group.add(shaft);
      billboards.push(shaft);
    }
  }

  const sky = toTexture(gradient(preset.sky[0], preset.sky[1], 64), false);
  sky.magFilter = sky.minFilter = THREE.LinearFilter;

  return {
    group,
    crowd,
    billboards,
    finishTop: new THREE.Vector3(postAt.x, 7, postAt.y),
    sky,
  };
}

/** Disposes every geometry, material and texture under `root`. */
export function disposeTree(root: THREE.Object3D) {
  const textures = new Set<THREE.Texture>();
  const materials = new Set<THREE.Material>();
  root.traverse((o) => {
    const mesh = o as THREE.Mesh;
    mesh.geometry?.dispose();
    const mats = mesh.material ? ([] as THREE.Material[]).concat(mesh.material) : [];
    if (mesh.customDepthMaterial) mats.push(mesh.customDepthMaterial);
    for (const m of mats) {
      materials.add(m);
      const map = (m as THREE.MeshBasicMaterial).map;
      if (map) textures.add(map);
    }
  });
  for (const m of materials) m.dispose();
  for (const t of textures) t.dispose();
}
```

```ts
import { ShaderPass } from 'three/addons/postprocessing/ShaderPass.js';
import { UnrealBloomPass } from 'three/addons/postprocessing/UnrealBloomPass.js';
import { HorizontalTiltShiftShader } from 'three/addons/shaders/HorizontalTiltShiftShader.js';
import { VerticalTiltShiftShader } from 'three/addons/shaders/VerticalTiltShiftShader.js';
import type { Preset } from './lighting.ts';
import { TILT_FOCUS } from './look.ts';

/** Colour grade + vignette in one pass (Look target 5). */
const GradeShader = {
  uniforms: {
    tDiffuse: { value: null },
    tint: { value: new THREE.Vector3(1, 1, 1) },
    saturation: { value: 1 },
    vignette: { value: 0.5 },
  },
  vertexShader: /* glsl */ `
    varying vec2 vUv;
    void main() {
      vUv = uv;
      gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
    }`,
  fragmentShader: /* glsl */ `
    uniform sampler2D tDiffuse;
    uniform vec3 tint;
    uniform float saturation;
    uniform float vignette;
    varying vec2 vUv;
    void main() {
      vec4 c = texture2D(tDiffuse, vUv);
      vec3 col = c.rgb * tint;
      float l = dot(col, vec3(0.2126, 0.7152, 0.0722));
      col = mix(vec3(l), col, saturation);
      float d = length(vUv - 0.5) * 1.2;
      col *= 1.0 - vignette * smoothstep(0.25, 0.75, d);
      gl_FragColor = vec4(col, c.a);
    }`,
};

export interface Post {
  readonly composer: EffectComposer;
  setSize(width: number, height: number): void;
  /** Blur in screen px at the band edges. */
  setTilt(blurPx: number): void;
  dispose(): void;
}

export function buildPost(
  renderer: THREE.WebGLRenderer,
  scene: THREE.Scene,
  camera: THREE.Camera,
  preset: Preset,
): Post {
  const composer = new EffectComposer(renderer);
  composer.addPass(new RenderPass(scene, camera));
  const tiltH = new ShaderPass(HorizontalTiltShiftShader);
  const tiltV = new ShaderPass(VerticalTiltShiftShader);
  tiltH.uniforms.r!.value = TILT_FOCUS;
  tiltV.uniforms.r!.value = TILT_FOCUS;
  composer.addPass(tiltH);
  composer.addPass(tiltV);
  const { strength, radius, threshold } = preset.bloom;
  composer.addPass(new UnrealBloomPass(new THREE.Vector2(256, 256), strength, radius, threshold));
  const grade = new ShaderPass(GradeShader);
  grade.uniforms.tint!.value = new THREE.Vector3(...preset.grade.tint);
  grade.uniforms.saturation!.value = preset.grade.saturation;
  grade.uniforms.vignette!.value = preset.grade.vignette;
  composer.addPass(grade);
  composer.addPass(new OutputPass());

  let w = 1;
  let h = 1;
  let blur = 2;
  const apply = () => {
    tiltH.uniforms.h!.value = blur / w;
    tiltV.uniforms.v!.value = blur / h;
  };
  return {
    composer,
    setSize(width, height) {
      w = width;
      h = height;
      composer.setSize(width, height);
      apply();
    },
    setTilt(blurPx) {
      blur = blurPx;
      apply();
    },
    dispose() {
      composer.dispose();
    },
  };
}
```

```ts
import * as THREE from 'three';
import { progressAt } from '../../sim/playback.ts';
import type { HorseId, Timeline } from '../../sim/types.ts';
import { COAST_SPEED, defaultColors, type ColorsOf } from '../Pixel2DRenderer.ts';
import type { RaceRenderer } from '../renderer.ts';
import { poseOf } from '../sprites.ts';
import { horsePoint, type Track } from '../track.ts';
import { presetById, presetOf, type Preset, type PresetId } from './lighting.ts';
import { FOV_DEG, PIXEL_M, TILT_BLUR_PX, distanceForScale } from './look.ts';
import { buildPost, type Post } from './post.ts';
import { buildWorld, cutoutMaterials, disposeTree, type World } from './scene.ts';
import { CELL, cellOf, cellUv, drawSheet, sheetSize } from './sheet.ts';

export interface HD2DOptions {
  /** Called once if the WebGL context is lost; RaceView swaps to 2D. */
  onLost?: () => void;
  /** Look test: hold race time here instead of following the clock. */
  freezeT?: number;
  /** Look test / sandbox: force a lighting preset. */
  preset?: PresetId;
}

interface HorseView {
  readonly horse: HorseId;
  readonly lane: number;
  readonly mesh: THREE.Mesh;
  readonly map: THREE.Texture;
}

const CHASE_PITCH = (35 * Math.PI) / 180;

export class HD2DRenderer implements RaceRenderer {
  private readonly canvas: HTMLCanvasElement;
  private readonly colors: ColorsOf;
  private readonly opts: HD2DOptions;
  private raf = 0;
  private renderer: THREE.WebGLRenderer | null = null;
  private scene: THREE.Scene | null = null;
  private camera: THREE.PerspectiveCamera | null = null;
  private post: Post | null = null;
  private world: World | null = null;
  private sheet: THREE.CanvasTexture | null = null;
  private horses: HorseView[] = [];
  private sun: THREE.DirectionalLight | null = null;
  private preset: Preset | null = null;
  private resize: ResizeObserver | null = null;
  private readonly lost = (e: Event) => {
    e.preventDefault();
    this.stop();
    this.opts.onLost?.();
  };

  constructor(canvas: HTMLCanvasElement, colors: ColorsOf = defaultColors, opts: HD2DOptions = {}) {
    this.canvas = canvas;
    this.colors = colors;
    this.opts = opts;
  }

  play(timeline: Timeline, track: Track, startAtMs: number) {
    this.stop();
    const lanes = timeline.horses.length;
    const preset = this.opts.preset ? presetById(this.opts.preset) : presetOf(timeline.seed);
    this.preset = preset;

    const renderer = new THREE.WebGLRenderer({
      canvas: this.canvas,
      antialias: false,
      powerPreference: 'high-performance',
    });
    renderer.setPixelRatio(1);
    renderer.shadowMap.enabled = true;
    renderer.shadowMap.type = THREE.PCFSoftShadowMap;
    renderer.toneMapping = THREE.ACESFilmicToneMapping;
    this.renderer = renderer;
    this.canvas.addEventListener('webglcontextlost', this.lost);

    const scene = new THREE.Scene();
    this.scene = scene;
    const world = buildWorld(track, lanes, timeline.seed, preset);
    this.world = world;
    scene.add(world.group);
    scene.background = world.sky;
    scene.fog = new THREE.Fog(preset.fog, 100, 400);
    scene.add(new THREE.HemisphereLight(preset.hemi.sky, preset.hemi.ground, preset.hemi.intensity));
    const sun = new THREE.DirectionalLight(preset.sun.color, preset.sun.intensity);
    sun.castShadow = true;
    sun.shadow.mapSize.set(1024, 1024);
    sun.shadow.bias = -0.0005;
    scene.add(sun, sun.target);
    this.sun = sun;

    // One sprite sheet for every lane; each horse gets a texture view with its own UV window.
    const { width, height } = sheetSize(lanes);
    const sheetCanvas = document.createElement('canvas');
    sheetCanvas.width = width;
    sheetCanvas.height = height;
    drawSheet(
      sheetCanvas.getContext('2d')!,
      timeline.horses.map((h, lane) => this.colors(h, lane)),
    );
    const sheet = new THREE.CanvasTexture(sheetCanvas);
    sheet.magFilter = THREE.NearestFilter;
    sheet.minFilter = THREE.NearestFilter;
    sheet.generateMipmaps = false;
    sheet.colorSpace = THREE.SRGBColorSpace;
    this.sheet = sheet;
    const geo = new THREE.PlaneGeometry(CELL.w * PIXEL_M, CELL.h * PIXEL_M);
    geo.translate(0, (CELL.footY - CELL.h / 2) * PIXEL_M, 0);
    this.horses = timeline.horses.map((horse, lane) => {
      const map = sheet.clone();
      const { material, depth } = cutoutMaterials(map);
      const mesh = new THREE.Mesh(geo, material);
      mesh.customDepthMaterial = depth;
      mesh.castShadow = true;
      scene.add(mesh);
      return { horse, lane, mesh, map };
    });

    const camera = new THREE.PerspectiveCamera(FOV_DEG, 16 / 9, 1, 3000);
    this.camera = camera;
    this.post = buildPost(renderer, scene, camera, preset);
    this.post.setTilt(TILT_BLUR_PX.chase);

    const fit = () => {
      const w = Math.max(1, this.canvas.clientWidth);
      const h = Math.max(1, Math.round((w * 9) / 16));
      renderer.setSize(w, h, false);
      this.post?.setSize(w, h);
      camera.aspect = w / h;
      camera.updateProjectionMatrix();
    };
    fit();
    this.resize = new ResizeObserver(fit);
    this.resize.observe(this.canvas);

    let last = performance.now();
    const frame = () => {
      const now = performance.now();
      if (import.meta.env.DEV) {
        // Dev-only readout for /sandbox (spec §4 performance target).
        const g = globalThis as { hd2dFrameMs?: number };
        g.hd2dFrameMs = (g.hd2dFrameMs ?? 16) * 0.9 + (now - last) * 0.1;
      }
      last = now;
      const t = this.opts.freezeT ?? (Date.now() - startAtMs) / 1000;
      this.draw(timeline, track, t);
      this.raf = requestAnimationFrame(frame);
    };
    frame();
  }

  /** Metres along the race, coasting past the line after the finish (display only). */
  private metres(timeline: Timeline, horse: HorseId, t: number) {
    const finished = timeline.finishTimes[horse] ?? Infinity;
    return progressAt(timeline, horse, t) + Math.max(0, t - finished) * COAST_SPEED;
  }

  private draw(timeline: Timeline, track: Track, t: number) {
    const { camera, post, sun, world, preset } = this;
    if (!camera || !post || !sun || !world || !preset) return;
    const d = timeline.distance;

    // Look-test camera: chase-style, centred on the top 3, from the infield side.
    const order = [...timeline.horses].sort(
      (a, b) => this.metres(timeline, b, t) - this.metres(timeline, a, t),
    );
    const pack = order
      .slice(0, 3)
      .map((h) => horsePoint(track, d, this.metres(timeline, h, t), timeline.horses.indexOf(h)));
    const tx = pack.reduce((s, p) => s + p.x, 0) / pack.length;
    const ty = pack.reduce((s, p) => s + p.y, 0) / pack.length;
    const lead = this.metres(timeline, order[0]!, t);
    const a = horsePoint(track, d, lead, 0);
    const b = horsePoint(track, d, lead, 1);
    const yaw = Math.atan2(-(b.x - a.x), -(b.y - a.y)); // inward normal: infield side
    const dist = distanceForScale();
    camera.position.set(
      tx + Math.sin(yaw) * Math.cos(CHASE_PITCH) * dist,
      Math.sin(CHASE_PITCH) * dist,
      ty + Math.cos(yaw) * Math.cos(CHASE_PITCH) * dist,
    );
    camera.lookAt(tx, 0, ty);
    const fog = this.scene!.fog as THREE.Fog;
    fog.near = dist * 1.2;
    fog.far = dist * 4;

    // Sun shadows follow the camera target.
    const [fx, fy, fz] = preset.sun.from;
    sun.target.position.set(tx, 0, ty);
    sun.position.set(tx + fx * 200, fy * 200, ty + fz * 200);
    const cam = sun.shadow.camera;
    cam.left = cam.bottom = -60;
    cam.right = cam.top = 60;
    cam.near = 1;
    cam.far = 500;
    cam.updateProjectionMatrix();

    // Everything flat faces the camera's yaw (parallel billboards).
    for (const o of world.billboards) o.rotation.y = yaw;

    // Horses: position, facing, pose and gallop frame.
    const p = new THREE.Vector3();
    const q = new THREE.Vector3();
    for (const v of this.horses) {
      const m = this.metres(timeline, v.horse, t);
      const here = horsePoint(track, d, m, v.lane);
      const ahead = horsePoint(track, d, m + 1, v.lane);
      v.mesh.position.set(here.x, 0, here.y);
      v.mesh.rotation.y = yaw;
      p.set(here.x, 0, here.y).project(camera);
      q.set(ahead.x, 0, ahead.y).project(camera);
      const pose = poseOf(q.x - p.x, -(q.y - p.y));
      const uv = cellUv(timeline.horses.length, cellOf(v.lane, pose, Math.floor(m / 1.5)));
      v.map.offset.set(uv.offsetX, uv.offsetY);
      v.map.repeat.set(uv.repeatX, uv.repeatY);
    }

    post.composer.render();
  }

  stop() {
    cancelAnimationFrame(this.raf);
    this.resize?.disconnect();
    this.resize = null;
    this.canvas.removeEventListener('webglcontextlost', this.lost);
    if (this.scene) disposeTree(this.scene);
    this.world?.sky.dispose();
    this.sheet?.dispose();
    for (const v of this.horses) v.map.dispose();
    this.post?.dispose();
    this.renderer?.dispose();
    this.renderer = null;
    this.scene = null;
    this.camera = null;
    this.post = null;
    this.world = null;
    this.sheet = null;
    this.sun = null;
    this.preset = null;
    this.horses = [];
  }
}
```

```tsx
export function Silk({ color }: { color: string }) {

type Props = {

export function RaceView({
  timeline,
  startAtMs,
  names,
  colors,
  track = TRACK,
  look,
  children,
}: Props) {
  const canvas = useRef<HTMLCanvasElement>(null);
  const box = useRef<HTMLDivElement>(null);

    document.addEventListener('fullscreenchange', sync);
    return () => document.removeEventListener('fullscreenchange', sync);

    (full ? document.exitFullscreen() : box.current!.requestFullscreen()).catch(() => {});

    colorsRef.current = colors ?? defaultColors;

    const el = canvas.current;
    if (!el) return;

    if (active === '2d') {
      const r = new Pixel2DRenderer(el, colorsOf);
      r.play(timeline, track, startAtMs);
      return () => r.stop();

    let r: RaceRenderer | null = null;
    let cancelled = false;

      console.warn('HD-2D unavailable, using the 2D view:', why);
      r?.stop();
      if (!cancelled) setHdFailed(true);
    };
    import('./hd2d/HD2DRenderer.ts')
      .then(({ HD2DRenderer }) => {
        if (cancelled) return;
        r = new HD2DRenderer(el, colorsOf, { ...look, onLost: () => fail('WebGL context lost') });
        r.play(timeline, track, startAtMs);
      })
      .catch(fail);
    return () => {
      cancelled = true;
      r?.stop();
    };
  }, [timeline, startAtMs, track, active, look]);

  const t = useRaceTime(startAtMs);
  const toggleMode = () => {
    const next: RenderMode = mode === 'hd' ? '2d' : 'hd';
    setMode(next);
    saveRenderMode(next);
  };
  return (
    <div className="race" ref={box}>
      <div className="race-buttons">
        {document.fullscreenEnabled && (
          <button type="button" className="chip" aria-pressed={full} onClick={toggleFull}>
            {full ? 'Exit fullscreen' : '⛶ Fullscreen'}
          </button>
        )}
        <button
          type="button"
          className="chip"
          aria-pressed={active === 'hd'}
          disabled={!webgl2 || hdFailed}
          onClick={toggleMode}
        >
          {webgl2 && !hdFailed ? 'HD' : 'HD unavailable'}
        </button>
      </div>
      <div className="panel race-frame">
        {/* A new element per renderer: a canvas with a WebGL context can't give a 2D one. */}
        <canvas
          key={active}
          ref={canvas}
          className={active === 'hd' ? 'hd' : undefined}
          width={active === '2d' ? VIEW.width : undefined}
          height={active === '2d' ? VIEW.height : undefined}
          role="img"
          aria-label={`${track.name} race, ${timeline.horses.length} horses (standings listed alongside)`}
        />
      </div>
      <p className="visually-hidden" role="status" aria-live="polite">
        {raceAnnouncement(timeline, t, names)}
      </p>
      <ol className="standings panel">
        {standingsAt(timeline, t).map((h) => {

              {time <= t && ` · ${time.toFixed(2)} s`}
            </li>
          );
        })}
      </ol>
      {children}
    </div>
  );
}
```

```ts
import { progressAt, standingsAt } from '../../sim/playback.ts';
import type { Timeline } from '../../sim/types.ts';
import { FINISH_SHOT_METRES } from '../camera.ts';
import { finishS, horsePoint, trackBounds, trackPoint, type Point, type Track } from '../track.ts';
import { FINISH_DISTANCE_SCALE, FOV_DEG, PITCH_DEG, distanceForScale } from './look.ts';

export type ShotKind = 'wide' | 'chase' | 'finish';

export interface Shot {
  readonly kind: ShotKind;
  /** Look-at point, track metres. */
  readonly x: number;
  readonly y: number;
  /** Radians; the camera sits at target + (sin yaw, ·, cos yaw) in Three.js x/z. */
  readonly yaw: number;
  /** Radians below horizontal. */
  readonly pitch: number;
  /** Metres from the target. */
  readonly distance: number;
}

export const WIDE_SECONDS = 2;
const ASPECT = 16 / 9;
const rad = (deg: number) => (deg * Math.PI) / 180;

/** Yaw that puts the camera on the infield side of the track at `s` (inward normal). */
function infieldYaw(track: Track, s: number) {
  const a = trackPoint(track, s, 0);
  const b = trackPoint(track, s, 1);
  return Math.atan2(-(b.x - a.x), -(b.y - a.y));
}

function wideShot(track: Track, lanes: number): Shot {
  const { halfWidth, halfHeight } = trackBounds(track, lanes);
  const pitch = rad(PITCH_DEG.wide);
  const tanHalf = Math.tan(rad(FOV_DEG) / 2);
  const distance =
    1.15 * Math.max(halfWidth / (ASPECT * tanHalf), (halfHeight * Math.sin(pitch)) / tanHalf);
  // From the infield side of the home straight (camera toward -y), like the other shots.
  return { kind: 'wide', x: 0, y: 0, yaw: Math.PI, pitch, distance };
}

/** Where the camera wants to be at race time `t`; the renderer eases within a shot, cuts between. */
export function shotAt(track: Track, timeline: Timeline, t: number, motion = true): Shot {
  const lanes = timeline.horses.length;
  if (!motion || t < WIDE_SECONDS) return wideShot(track, lanes);
  const { distance, horses } = timeline;
  const order = standingsAt(timeline, t);
  const lead = progressAt(timeline, order[0]!, t);
  if (lead >= distance - FINISH_SHOT_METRES) {
    const p = horsePoint(track, distance, distance, (lanes - 1) / 2);
    return {
      kind: 'finish',
      x: p.x,
      y: p.y,
      yaw: infieldYaw(track, finishS(track)),
      pitch: rad(PITCH_DEG.finish),
      distance: distanceForScale() * FINISH_DISTANCE_SCALE,
    };
  }
  const pack = order
    .slice(0, 3)
    .map((h) => horsePoint(track, distance, progressAt(timeline, h, t), horses.indexOf(h)));
  const mean = (f: (p: Point) => number) => pack.reduce((sum, p) => sum + f(p), 0) / pack.length;
  return {
    kind: 'chase',
    x: mean((p) => p.x),
    y: mean((p) => p.y),
    yaw: infieldYaw(track, finishS(track) - distance + lead),
    pitch: rad(PITCH_DEG.chase),
    distance: distanceForScale(),
  };
}

/** Camera position in Three.js coords (x, up, z). */
export function cameraPosition(shot: Shot) {
  const flat = Math.cos(shot.pitch) * shot.distance;
  return {
    x: shot.x + Math.sin(shot.yaw) * flat,
    y: Math.sin(shot.pitch) * shot.distance,
    z: shot.y + Math.cos(shot.yaw) * flat,
  };
}
```

```ts
import type { HorseId, RaceEvent } from '../../sim/types.ts';
import { ITEM_HOLD_SECONDS } from '../Pixel2DRenderer.ts';

export type EffectKind = 'speed' | 'hit' | 'shield' | 'reflect' | 'dud' | 'box';

export const EFFECT_SECONDS: Readonly<Record<EffectKind, number>> = {
  speed: 1.2,
  hit: 0.8,
  shield: 1,
  reflect: 0.6,
  dud: 1.5,
  box: 0.5,
};

export interface Effect {
  readonly kind: EffectKind;
  readonly horse: HorseId;
  /** Race time it starts. */
  readonly t0: number;
}

/** Built once per race. A box opens when it is grabbed on screen (ITEM_HOLD_SECONDS early). */
export function effectsOf(events: readonly RaceEvent[]): Effect[] {
  const out: Effect[] = [];
  for (const e of events) {
    if (e.type === 'pickup') {
      out.push({ kind: 'box', horse: e.horse, t0: e.t - ITEM_HOLD_SECONDS });
      if (e.item?.startsWith('speed:')) out.push({ kind: 'speed', horse: e.horse, t0: e.t });
      if (e.item?.startsWith('dud:')) out.push({ kind: 'dud', horse: e.horse, t0: e.t });
    } else if (e.type === 'hit') out.push({ kind: 'hit', horse: e.horse, t0: e.t });
    else if (e.type === 'blocked') out.push({ kind: 'shield', horse: e.horse, t0: e.t });
    else if (e.type === 'reflected') out.push({ kind: 'reflect', horse: e.horse, t0: e.t });
  }
  return out;
}

export function activeAt(effects: readonly Effect[], t: number) {
  return effects
    .map((effect) => ({ effect, age: t - effect.t0 }))
    .filter(({ effect, age }) => age >= 0 && age < EFFECT_SECONDS[effect.kind]);
}

/**
 * Horse-local metres: `ahead` along the running direction, `up`, `side` to the horse's left.
 * `color` is a CSS hex; the renderer multiplies it by `alpha` (additive blending fades it out).
 */
export interface Particle {
  readonly ahead: number;
  readonly up: number;
  readonly side: number;
  readonly color: string;
  readonly alpha: number;
}

export const PARTICLES = 12;

/** Deterministic 0..1 noise per particle index. */
const hash = (i: number, salt: number) => {
  const x = Math.sin(i * 12.9898 + salt * 78.233) * 43758.5453;
  return x - Math.floor(x);
};

export function particlesOf(kind: EffectKind, age: number): Particle[] {
  const u = Math.min(1, Math.max(0, age / EFFECT_SECONDS[kind]));
  const fade = 1 - u;
  return Array.from({ length: PARTICLES }, (_, i): Particle => {
    const h1 = hash(i, 1);
    const h2 = hash(i, 2);
    const ang = (i / PARTICLES) * 2 * Math.PI;
    switch (kind) {
      case 'speed': // streaks and dust flying back
        return {
          ahead: -(0.3 + (i % 4) * 0.4 + u * 2.5),
          up: 0.3 + h1 * 1.8,
          side: (h2 - 0.5) * 0.8,
          color: i % 3 ? '#ffd54f' : '#ffffff',
          alpha: fade,
        };
      case 'hit': // flash and stars bursting from the head
        return {
          ahead: Math.cos(ang) * u * 1.5,
          up: 1.8 + Math.sin(ang) * u * 1.5,
          side: (h1 - 0.5) * 0.4,
          color: i % 2 ? '#ffffff' : '#fff176',
          alpha: fade,
        };
      case 'shield': // a turning bubble
        return {
          ahead: Math.cos(ang + u * 3) * 1.4,
          up: 1.2 + Math.sin(ang + u * 3) * 1.4,
          side: Math.sin(2 * ang) * 0.3,
          color: '#81d4fa',
          alpha: fade * 0.8,
        };
      case 'reflect': // sparks shooting up
        return {
          ahead: (h1 - 0.5) * 1,
          up: 1.5 + u * 2 * (0.5 + h2),
          side: (hash(i, 3) - 0.5) * 1,
          color: '#e1f5fe',
          alpha: fade,
        };
      case 'dud': // dizzy swirl over the head, fading over its last fifth
        return {
          ahead: Math.cos(ang + u * 6) * 0.5,
          up: 2.6 + (i % 3) * 0.15,
          side: Math.sin(ang + u * 6) * 0.5,
          color: i % 2 ? '#ce93d8' : '#fff59d',
          alpha: u < 0.8 ? 1 : (1 - u) / 0.2,
        };
      case 'box': // a puff where the box was
        return {
          ahead: Math.cos(ang) * (0.3 + u * 1.2),
          up: 0.4 + u * 0.8,
          side: Math.sin(ang) * (0.3 + u * 1.2),
          color: '#ffffff',
          alpha: fade,
        };
    }
  });
}
```

```ts
import { SILK_COLORS } from '../palette.ts';

export const MOTES = 120;
export const CONFETTI = 160;
export const CONFETTI_SECONDS = 4;

const hash = (i: number, salt: number) => {
  const x = Math.sin(i * 12.9898 + salt * 78.233) * 43758.5453;
  return x - Math.floor(x);
};

/** A floating mote around the camera target (offsets in metres, y up from the ground). */
export function moteAt(t: number, kind: 'dust' | 'fireflies', i: number) {
  const x = (hash(i, 1) - 0.5) * 60 + Math.sin(t * 0.3 + i) * 1.5;
  const y = 1 + hash(i, 2) * 12 + Math.sin(t * 0.5 + i * 2) * 0.5;
  const z = (hash(i, 3) - 0.5) * 60 + Math.cos(t * 0.27 + i) * 1.5;
  const alpha = kind === 'fireflies' ? (Math.sin(t * 3 + i * 1.7) + 1) / 2 : 0.5;
  return { x, y, z, alpha };
}

/**
 * Confetti offsets from the top of the finish post, `age` seconds after the first finish.
 * `y` never drops below the start height (it lands on an invisible shelf, then fades).
 */
export function confettiAt(age: number, i: number) {
  const vx = (hash(i, 4) - 0.5) * 6;
  const vy = 6 + hash(i, 5) * 6;
  const vz = (hash(i, 6) - 0.5) * 6;
  const g = 4;
  const y = Math.max(0, vy * age - 0.5 * g * age * age);
  const alpha = age >= CONFETTI_SECONDS ? 0 : age < 3 ? 1 : 1 - (age - 3) / (CONFETTI_SECONDS - 3);
  return { x: vx * age, y, z: vz * age, alpha, color: SILK_COLORS[i % SILK_COLORS.length]! };
}
```
